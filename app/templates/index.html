async function handleLogin(event) { event.preventDefault(); const btn = document.getElementById('loginBtn'); const statusBox = document.getElementById('loginStatus'); const email = document.getElementById('email').value.trim(); const password = document.getElementById('password').value; const remember = document.getElementById('remember').checked; // Extract return_to from the current query string if present const urlParams = new URLSearchParams(window.location.search); const returnTo = urlParams.get('return_to'); btn.innerHTML = 'Verifying credentials...'; btn.disabled = true; statusBox.style.display = 'none'; const formData = new FormData(); formData.append('email', email); formData.append('password', password); formData.append('remember', remember); if (returnTo) { formData.append('return_to', returnTo); } try { const response = await fetch('/api/login', { method: 'POST', body: formData }); const data = await response.json(); if (response.ok) { statusBox.style.display = 'block'; statusBox.style.backgroundColor = 'rgba(0, 240, 255, 0.15)'; statusBox.style.border = '1px solid var(--accent-cyan)'; statusBox.style.color = 'var(--accent-cyan)'; statusBox.innerText = 'Authentication successful. Redirecting to workspace...'; setTimeout(() => { window.location.href = data.redirect || 'http://localhost:8020/'; }, 600); } else { throw new Error(data.detail || 'Invalid email or password.'); } } catch (err) { statusBox.style.display = 'block'; statusBox.style.backgroundColor = 'rgba(239, 68, 68, 0.15)'; statusBox.style.border = '1px solid rgba(239, 68, 68, 0.5)'; statusBox.style.color = '#FCA5A5'; statusBox.innerText = err.message; btn.innerHTML = 'Authenticate →'; btn.disabled = false; } }
